import { Link, useFetcher, useLoaderData } from "@kasperrt/smolter/preact";
import type { loader } from "./loader";
import type { loader as taskLoader } from "../tasks.$id/loader";
import type { action } from "./action";

export default function Tasks() {
  const data = useLoaderData<typeof loader>();
  const create = useFetcher<typeof action>();
  const preview = useFetcher<typeof taskLoader>();

  return <>
    {data?.tasks.map(task => (
      <Link key={task.id} href={`/tasks/${task.id}`}>
        {task.title}
      </Link>
    ))}

    <create.Form action="/tasks" method="post">
      <input name="title" required />
      <button disabled={create.state !== "idle"}>Create task</button>
    </create.Form>

    <button disabled={preview.state !== "idle"} onClick={() => preview.load("/tasks/1")}>
      Fetch task 1
    </button>
    <p>{preview.data?.title}</p>
  </>;
}
