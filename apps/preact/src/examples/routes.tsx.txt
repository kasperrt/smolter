// vite.config.ts
import { defineConfig } from "vite";
import { fileRouter } from "smolter/vite";

export default defineConfig({
  plugins: [fileRouter({ directory: "src/routes" })],
});

// Flat route folders → URL paths
// src/routes/_index/route.ts       → /
// src/routes/tasks._index/route.ts → /tasks
// src/routes/tasks.$id/route.ts    → /tasks/:id
// src/routes/about/route.ts        → /about

// main.ts
import routes from "virtual:smolter/routes";
import { createRouter, RouterProvider, Routes } from "smolter/preact";

const router = createRouter(routes, { browser: true });

// Inside your app
<RouterProvider router={router}>
  <Routes />
</RouterProvider>
