import { useFetcher, useLoaderData, useRevalidator } from "@kasperrt/smolter/react";
import type { clientLoader } from "./loader";
import type { clientAction } from "./action";

export default function Counter() {
  const data = useLoaderData<typeof clientLoader>();
  const fetcher = useFetcher<typeof clientAction>();
  const revalidate = useRevalidator();
  const busy = fetcher.state !== "idle";

  return <>
    <output>{data?.count}</output>

    <fetcher.Form action="/" method="post">
      <button disabled={busy} name="intent" value="increment">Add one +</button>
      <button disabled={busy} name="intent" value="reset">Reset</button>
    </fetcher.Form>

    <p>{fetcher.error?.message ?? fetcher.data?.message}</p>
    <p>Loader runs: {data?.reads}</p>

    <button onClick={() => revalidate()}>Revalidate ↻</button>
  </>;
}
